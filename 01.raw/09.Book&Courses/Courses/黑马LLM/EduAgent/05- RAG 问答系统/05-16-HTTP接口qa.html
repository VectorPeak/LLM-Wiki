    5.16 HTTP 接口（qa.py）¶

    学习目标¶

    • 掌握两种对话接口的设计差异：普通同步接口 vs SSE 流式接口

    • 理解 on_chain_end 多次触发的问题及防御写法

    5.16.1 文件结构与 Pydantic 模型¶

    # backend/api/v1/qa.py

import json
from fastapi import APIRouter, Depends, HTTPException, status
from pydantic import BaseModel, Field
from sse_starlette.sse import EventSourceResponse
from langchain_core.messages import HumanMessage

from backend.agents.qa.graph import build_qa_graph
from backend.core.memory import build_thread_id
from backend.dependencies import get_current_user
from backend.core.logger import get_logger

router = APIRouter()
logger = get_logger(__name__)


# ── 请求 / 响应模型 ───────────────────────────────────────────

class ChatRequest(BaseModel):
    session_id:        str        = Field(..., description="会话 ID")
    course_id:         str | None = Field(None, description="课程 ID（可选，限定检索范围）")
    message:           str        = Field(..., min_length=1, max_length=2000)
    enable_web_search: bool       = Field(False, description="低置信度时是否先走 Web Search 再给 LLM")


class ChatResponse(BaseModel):
    session_id:    str
    answer:        str
    answer_mode:   str        # "rag" / "web_augmented" / "llm_direct" / "general"
    confidence:    float
    sources:       list[str]
    fallback_used: bool


class SessionMessage(BaseModel):
    role:       str   # "user" / "assistant"
    content:    str
    created_at: str


class HistoryResponse(BaseModel):
    session_id:  str
    messages:    list[SessionMessage]
    summary:     str | None
    total_turns: int

    为什么这里导入 build_qa_graph 而不是 Orchestrator？

    Orchestrator 是第 8 章实现的统一路由层，负责把请求分发到不同 Agent。第 5 章只需要 QA Agent，直接调图即可——这既是完整功能，也是最直接的实现。第 8 章重构时，/chat 接口会改为走 Orchestrator，/chat/stream 和历史接口保持直接调图不变（流式场景 Orchestrator 并不介入）。

    5.16.2 POST /chat（同步接口）¶

    @router.post("/chat", response_model=ChatResponse)
async def chat(
    req: ChatRequest,
    current_user: dict = Depends(get_current_user),
):
    """智能问答：发送消息，获取 RAG 或 LLM 直答（非流式）"""
    graph     = build_qa_graph()
    thread_id = build_thread_id(current_user["user_id"], req.session_id)

    initial_state = {
        "messages":            [HumanMessage(content=req.message)],
        "student_id":          current_user["user_id"],
        "tenant_id":           current_user["tenant_id"],
        "session_id":          req.session_id,
        "course_id":           req.course_id,
        "query_type":          "PRECISE",     # 占位初始值，classify_query_node 内部会动态覆盖
        "enable_web_search":   req.enable_web_search,
        "web_search_results":  [],            # 每轮重置，防止上轮搜索结果污染本轮 sources
    }
    config: dict = {"configurable": {"thread_id": thread_id}}

    try:
        result = await graph.ainvoke(initial_state, config=config)
    except Exception as e:
        logger.error("chat.invoke_error", error=str(e), exc_info=True)
        raise HTTPException(
            status_code=status.HTTP_500_INTERNAL_SERVER_ERROR,
            detail={"code": "AGENT_ERROR", "message": str(e)},
        )

    return ChatResponse(
        session_id=req.session_id,
        answer=result.get("answer", ""),
        answer_mode=result.get("answer_mode", "llm_direct"),
        confidence=result.get("confidence", 0.0),
        sources=result.get("sources", []),
        fallback_used=result.get("fallback_used", False),
    )

    query_type="PRECISE" 只是占位：classify_query_node 是图的第一个节点，运行后会把 query_type 覆盖为实际分类结果（GENERAL / PRECISE / VAGUE / BROAD）。这里写 "PRECISE" 是为了让 QAState 字段有初始值，不影响最终分类。

    **graph.ainvoke() 返回值**就是最终 QAState 的 dict。QAState 里有 answer、answer_mode、confidence、sources、fallback_used 字段，直接读取即可，无需额外解包。

    5.16.3 POST /chat/stream（SSE 流式接口）¶

    流式接口直接拿图调用 astream_events，实时把 LLM token 推给前端。

    SSE 事件格式¶

    data: {"type": "progress", "stage": "理解问题中..."}    ← 节点开始时
data: {"type": "token",    "content": "Spring"}         ← LLM 实时 token
data: {"type": "token",    "content": " IOC"}
data: {"type": "meta",     "answer_mode": "rag",
        "confidence": 0.87, "sources": [...],
        "session_id": "..."}                             ← 流结束后的完整元数据
data: {"type": "done"}                                   ← 结束信号
data: {"type": "error",    "message": "..."}             ← 出错时替代以上全部

    前端用 fetch + ReadableStream 逐行解析（EventSource 不支持 POST）：

    const resp = await fetch("/api/v1/qa/chat/stream", {
    method: "POST",
    headers: {
        "Authorization": `Bearer ${token}`,
        "Content-Type": "application/json"
    },
    body: JSON.stringify({ session_id, message }),
});
const reader = resp.body.getReader();
// 逐行解析 "data: {...}\n\n" 格式

    完整实现¶

    @router.post("/chat/stream")
async def chat_stream(
    req: ChatRequest,
    current_user: dict = Depends(get_current_user),
):
    """智能问答流式接口（SSE），与 /chat 入参相同，响应改为流式推送"""
    graph     = build_qa_graph()
    thread_id = build_thread_id(current_user["user_id"], req.session_id)

    initial_state = {
        "messages":            [HumanMessage(content=req.message)],
        "student_id":          current_user["user_id"],
        "tenant_id":           current_user["tenant_id"],
        "session_id":          req.session_id,
        "course_id":           req.course_id,
        "query_type":          "PRECISE",
        "enable_web_search":   req.enable_web_search,
        "web_search_results":  [],            # 每轮重置，防止上轮搜索结果污染本轮 sources
    }
    config: dict = {"configurable": {"thread_id": thread_id}}

    # 只对这三个节点做 token 级流式推送
    _GENERATE_NODES = {"generate_rag", "generate_direct", "generate_general"}

    # 节点开始时推送的进度文案
    _PROGRESS_LABELS = {
        "classify_query":      "理解问题中...",
        "hyde_generate":       "理解问题中...",
        "multi_query_rewrite": "改写查询中...",
        "retrieve":            "召回相关文档...",
        "web_search":          "搜索互联网...",
        "generate_general":    "思考中...",
    }

    async def event_generator():
        answer_mode = "llm_direct"
        confidence  = 0.0
        sources: list[str] = []

        try:
            async for event in graph.astream_events(
                initial_state, config=config, version="v2"
            ):
                evt  = event["event"]
                node = event.get("metadata", {}).get("langgraph_node", "")

                # ── 节点开始：推送进度提示 ──────────────────────
                if evt == "on_chain_start" and node in _PROGRESS_LABELS:
                    yield {
                        "data": json.dumps(
                            {"type": "progress", "stage": _PROGRESS_LABELS[node]},
                            ensure_ascii=False,
                        )
                    }

                # ── LLM 实时 token ──────────────────────────────
                elif evt == "on_chat_model_stream" and node in _GENERATE_NODES:
                    chunk = event["data"].get("chunk")
                    if chunk and chunk.content:
                        yield {
                            "data": json.dumps(
                                {"type": "token", "content": chunk.content},
                                ensure_ascii=False,
                            )
                        }

                # ── 节点结束：捕获元数据 ────────────────────────
                elif evt == "on_chain_end" and node in _GENERATE_NODES:
                    output = event["data"].get("output", {})
                    if isinstance(output, dict):
                        _mode = output.get("answer_mode")
                        if _mode:
                            answer_mode = _mode
                        _srcs = output.get("sources")
                        if _srcs is not None:
                            sources = _srcs
                        _conf = (output.get("structured_output") or {}).get("confidence")
                        if _conf is not None:
                            confidence = _conf

        except Exception as e:
            logger.error("chat_stream.error", error=str(e), exc_info=True)
            yield {
                "data": json.dumps(
                    {"type": "error", "message": "流式输出异常，请使用普通接口重试"},
                    ensure_ascii=False,
                )
            }
            return

        # ── 元数据帧（流结束后一次性推送）──────────────────────
        yield {
            "data": json.dumps(
                {
                    "type":        "meta",
                    "session_id":  req.session_id,
                    "answer_mode": answer_mode,
                    "confidence":  confidence,
                    "sources":     sources,
                },
                ensure_ascii=False,
            )
        }
        yield {"data": json.dumps({"type": "done"})}

    return EventSourceResponse(event_generator())

    on_chain_end 为什么要做防御判断¶

    LangGraph 对同一节点会触发**多次** on_chain_end——节点内部的 LLM chain 结束时也会触发一次，那次 output 是 LLM 的原始输出（AIMessage 对象，不是 dict），不包含我们关心的 answer_mode / sources 字段。

    防御写法： 

    _mode = output.get("answer_mode")
if _mode:          # 只在字段明确存在时才更新
    answer_mode = _mode

    如果写成 answer_mode = output.get("answer_mode", answer_mode)，内层事件触发时 output.get("answer_mode") 返回 None，None or answer_mode 虽然安全，但不如显式 if _mode 意图清晰。

    5.16.4 GET /sessions/{session_id}/history¶

    两个数据源： - DB（qa_sessions 表）→ 历史摘要文本 - MemorySaver（graph.aget_state()）→ 消息列表

    @router.get("/sessions/{session_id}/history", response_model=HistoryResponse)
async def get_session_history(
    session_id: str,
    current_user: dict = Depends(get_current_user),
):
    """获取会话历史消息与摘要"""
    from sqlalchemy import text as sa_text
    from langchain_core.messages import HumanMessage as LCHuman, AIMessage as LCAi
    from backend.dependencies import AsyncSessionLocal

    student_id = current_user["user_id"]
    thread_id  = build_thread_id(student_id, session_id)

    # ── ① 从 DB 读摘要 ────────────────────────────────────────
    summary = None
    try:
        async with AsyncSessionLocal() as db_session:
            result = await db_session.execute(
                sa_text(
                    "SELECT summary FROM qa_sessions "
                    "WHERE thread_id = :tid AND student_id = :sid"
                ),
                {"tid": thread_id, "sid": student_id},
            )
            row = result.fetchone()
            if row:
                summary = row[0]
    except Exception as e:
        logger.warning("get_history.db_error", error=str(e))

    # ── ② 从 MemorySaver 读消息历史 ──────────────────────────
    messages: list[SessionMessage] = []
    total_turns = 0
    try:
        graph  = build_qa_graph()
        config = {"configurable": {"thread_id": thread_id}}
        state  = await graph.aget_state(config)
        if state and state.values:
            for msg in state.values.get("messages", []):
                content = (
                    msg.text
                    if hasattr(msg, "text") and not callable(msg.text)
                    else str(msg.content)
                )
                if isinstance(msg, LCHuman):
                    messages.append(SessionMessage(role="user", content=content, created_at=""))
                elif isinstance(msg, LCAi):
                    messages.append(SessionMessage(role="assistant", content=content, created_at=""))
            total_turns = sum(1 for m in messages if m.role == "user")
    except Exception as e:
        logger.warning("get_history.checkpoint_error", error=str(e))

    return HistoryResponse(
        session_id=session_id,
        messages=messages,
        summary=summary,
        total_turns=total_turns,
    )

    为什么摘要存 DB、消息历史存 MemorySaver？

    • MemorySaver 是进程内存，重启后消失，适合存短期对话消息（丢了重新问即可）
    • 摘要是对长期学习历程的压缩，需要持久化，存 DB 保证重启后仍然可用
    • 两者各司其职，classify_query_node 每轮从 DB 读摘要注入 Prompt，消息历史由 MemorySaver 自动续接

    章节总结¶

    • 三个接口全部直接调图：build_qa_graph() 返回已编译的 CompiledStateGraph，ainvoke / astream_events / aget_state 均可直接调用
    • /chat：graph.ainvoke() 返回最终 State dict，直接读 answer / answer_mode / confidence / sources / fallback_used
    • /chat/stream：astream_events 流式推送，四类事件：progress → token → meta → done
    • on_chain_end 防御：同一节点多次触发，只在字段明确存在时更新
    • 两数据源：摘要在 DB（持久），消息在 MemorySaver（进程内），各自最适合的存储位置

    第 8 章说明：Orchestrator 实现后，/chat 接口会重构为通过 get_orchestrator().handle() 调用，统一错误处理和 Agent 路由。/chat/stream 和 /history 接口保持直接调图不变。
