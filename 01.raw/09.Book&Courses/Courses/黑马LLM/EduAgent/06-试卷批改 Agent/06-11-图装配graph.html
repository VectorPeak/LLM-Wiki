    6.11 图装配（graph.py）¶

    学习目标¶

    • 掌握线性图的装配方式：9个节点全部用固定边连接
    • 理解 compile(checkpointer=...) 对 HitL 的意义
    • 验证图装配和节点注册是否正确

    💡 本节新建 graph.py，是 Exam Agent 的最后一个 Python 文件。

    📁 本节要创建的文件¶

    文件 状态 说明
    backend/agents/exam/graph.py 🆕 新建 图装配与编译

    6.11.1 图结构¶

    Exam Agent 是一条线性链，没有条件分支：

    START
  │
  ▼
parse_word
  │
  ▼
load_questions_meta
  │
  ▼
run_three_tracks
  │
  ▼
aggregate_results
  │
  ▼
analyze_weak_points
  │
  ▼
notify_teacher
  │
  ▼
teacher_review  ←── interrupt() 暂停点
  │
  ▼
apply_teacher_decision
  │
  ▼
publish_results
  │
  ▼
END

    与 QA Agent 不同，这里没有条件边——图的路径完全确定，每个节点只有一个出口。控制流的复杂性体现在节点内部（三轨并行、interrupt 暂停），而不是图的拓扑结构上。

    6.11.2 完整实现¶

    # backend/agents/exam/graph.py

from langgraph.graph import StateGraph, START, END

from backend.agents.exam.state import ExamState
from backend.agents.exam.nodes import (
    parse_word_node,
    load_questions_meta_node,
    run_three_tracks_node,
    aggregate_results_node,
    analyze_weak_points_node,
    notify_teacher_node,
    teacher_review_node,
    apply_teacher_decision_node,
    publish_results_node,
)
from backend.core.memory import get_memory_saver


def build_exam_graph():
    """
    构建并编译试卷批改 Agent 的 LangGraph 状态图。

    执行链路（线性）：
        parse_word → load_questions_meta → run_three_tracks
        → aggregate_results → analyze_weak_points
        → notify_teacher → teacher_review [interrupt]
        → apply_teacher_decision → publish_results → END
    """
    builder = StateGraph(ExamState)

    # ── 注册节点 ──────────────────────────────────────────────
    builder.add_node("parse_word",             parse_word_node)
    builder.add_node("load_questions_meta",    load_questions_meta_node)
    builder.add_node("run_three_tracks",       run_three_tracks_node)
    builder.add_node("aggregate_results",      aggregate_results_node)
    builder.add_node("analyze_weak_points",    analyze_weak_points_node)
    builder.add_node("notify_teacher",         notify_teacher_node)
    builder.add_node("teacher_review",         teacher_review_node)
    builder.add_node("apply_teacher_decision", apply_teacher_decision_node)
    builder.add_node("publish_results",        publish_results_node)

    # ── 固定边（线性链）──────────────────────────────────────
    builder.add_edge(START,                    "parse_word")
    builder.add_edge("parse_word",             "load_questions_meta")
    builder.add_edge("load_questions_meta",    "run_three_tracks")
    builder.add_edge("run_three_tracks",       "aggregate_results")
    builder.add_edge("aggregate_results",      "analyze_weak_points")
    builder.add_edge("analyze_weak_points",    "notify_teacher")
    builder.add_edge("notify_teacher",         "teacher_review")
    builder.add_edge("teacher_review",         "apply_teacher_decision")
    builder.add_edge("apply_teacher_decision", "publish_results")
    builder.add_edge("publish_results",        END)

    # ── 编译，绑定 MemorySaver ────────────────────────────────
    checkpointer = get_memory_saver("exam")
    return builder.compile(checkpointer=checkpointer)

    为什么 HitL 必须绑定 MemorySaver？

    interrupt() 把图"冻结"后，State 必须持久化到某个地方，等教师确认后才能恢复。如果不传 checkpointer，图是无状态的，interrupt() 后 State 丢失，无法恢复执行。MemorySaver 按 thread_id 把 State 存在进程内存中，Command(resume=...) 时按同一 thread_id 读取恢复。

    ✅ 验证：完整图端到端测试¶

    光验证节点列表能打印出来没有意义。本节直接跑完整图，验证从 parse_word 到 publish_results 的全链路，包括 HitL 的 interrupt 暂停与 Command(resume=) 恢复。

    6.11.3 测试脚本¶

    # 在项目根目录下运行：
# conda activate edu_agent
# python backend/agents/exam/graph.py

import sys, asyncio, uuid
sys.path.insert(0, ".")

from dotenv import load_dotenv
load_dotenv(".env.local")

from sqlalchemy import text
from langgraph.types import Command
from backend.agents.exam.graph import build_exam_graph
from backend.dependencies import AsyncSessionLocal

graph    = build_exam_graph()
EXAM_ID  = "e0000001-0000-0000-0000-000000000001"


async def main():
    # ── 测试准备：获取真实 student UUID，写入临时提交记录 ────────
    # 原因①：exam_reviews.submission_id 有 FK 约束，必须先在
    #         exam_submissions 中存在对应记录，否则 INSERT 报 ForeignKeyViolationError
    # 原因②：exam_submissions.student_id 有 FK 约束，必须用
    #         users 表中真实存在的 UUID，不能传 "test-student" 这种字符串
    async with AsyncSessionLocal() as session:
        row = await session.execute(
            text("SELECT id FROM users WHERE username = 'student01' LIMIT 1")
        )
        student_uuid = str(row.scalar())

    submission_id = str(uuid.uuid4())

    async with AsyncSessionLocal() as session:
        async with session.begin():
            # 清除旧测试记录，避免 UNIQUE(exam_id, student_id) 重跑冲突
            await session.execute(
                text("""
                    DELETE FROM exam_reviews
                    WHERE submission_id IN (
                        SELECT id FROM exam_submissions
                        WHERE exam_id = :exam_id AND student_id = :student_id
                    )
                """),
                {"exam_id": EXAM_ID, "student_id": student_uuid},
            )
            await session.execute(
                text("DELETE FROM exam_submissions WHERE exam_id=:exam_id AND student_id=:student_id"),
                {"exam_id": EXAM_ID, "student_id": student_uuid},
            )
            await session.execute(
                text("""
                    INSERT INTO exam_submissions (id, exam_id, student_id, tenant_id, status)
                    VALUES (:id, :exam_id, :student_id, 'tenant_default', 'processing')
                """),
                {"id": submission_id, "exam_id": EXAM_ID, "student_id": student_uuid},
            )

    # ── thread_id：MemorySaver 用它区分不同的批改任务 ─────────────
    # 两次 ainvoke() 必须用同一个 thread_id，图才能在第二次找回冻结的 State
    thread_id = f"test-{uuid.uuid4()}"
    config    = {"configurable": {"thread_id": thread_id}}

    initial_state = {
        "messages":        [],
        "word_file_path":  "./backend/agents/exam/student_answer.docx",
        "exam_id":         EXAM_ID,
        "submission_id":   submission_id,
        "student_id":      student_uuid,
        "tenant_id":       "tenant_default",
        "session_id":      "test-session",
        "teacher_decision": None,
        "fallback_used":   False,
        "sandbox_skipped": False,
    }

    # ══ 第一次 ainvoke：跑到 teacher_review_node 的 interrupt() 自动暂停 ══
    print("=" * 60)
    print("第一次 ainvoke：启动图，等待 interrupt 暂停...")
    print("=" * 60)
    result1 = await graph.ainvoke(initial_state, config=config)

    # interrupt() 暂停后，ainvoke 提前返回，State 已存入 MemorySaver
    # result1 包含截止到暂停时刻的所有 state 字段
    summary = result1.get("pre_review_summary", {})
    print(f"\n图已暂停（interrupt 触发）")
    print(f"AI 预评总分：{summary.get('total_score')} / {summary.get('full_score')}")
    print(f"需复核题数：{summary.get('needs_review_count')}")
    print(f"薄弱点数量：{len(result1.get('weak_points', []))}")
    print(f"\n>>> 此刻图冻结，等待教师决策 <<<\n")

    # ══ 模拟教师操作：构造 decision，第二次 ainvoke 恢复图 ══════
    decision = {
        "action":        "approve",   # 或 "modify"，见说明②
        "modifications": [],
        "teacher_id":    str(uuid.uuid4()),  # reviewed_by 列是 UUID 类型，必须合法
    }

    print("=" * 60)
    print("第二次 ainvoke：Command(resume=decision)，恢复图执行...")
    print("=" * 60)
    result2 = await graph.ainvoke(Command(resume=decision), config=config)

    final = result2.get("final_results", [])
    print(f"\n图执行完毕")
    print(f"最终批改题数：{len(final)}")
    for q in final:
        print(f"  题{q['question_no']} [{q['question_type']}] "
              f"最终得分 {q['final_score']}/{q['full_score']}")


asyncio.run(main())

    6.11.4 脚本中的两个关键设计¶

    ① 为什么需要测试准备步骤（DB 预插入）

    exam_reviews 表对 submission_id 有外键约束（exam_reviews_submission_id_fkey），exam_submissions 表对 student_id 有外键约束（REFERENCES users(id)）。

    如果直接用随机 UUID 和 "test-student" 这样的占位字符串，publish_results_node 的 INSERT 会报两种错误： - ForeignKeyViolationError：submission_id 不在 exam_submissions 中 - DataError：student_id / reviewed_by 不是合法 UUID 格式

    测试准备步骤做三件事： 1. 从 users 表查出 student01 的真实 UUID 2. 清除该学生在本试卷的旧记录（解决 UNIQUE(exam_id, student_id) 重跑冲突） 3. INSERT 一条 status='processing' 的提交记录，建立 FK 所需的父记录

    exam_reviews 先于 exam_submissions 删除，是因为 FK 约束方向：子表（exam_reviews）必须先删，才能删父表（exam_submissions）。

    ② 模拟"教师修改分数"

    把 decision 改成：

    decision = {
    "action":        "modify",
    "modifications": [
        {
            "question_id": "从 result1['pre_review_summary']['by_question'][0]['question_id'] 取",
            "new_score":   5,
            "comment":     "答案思路正确但表述不完整",
        }
    ],
    "teacher_id": str(uuid.uuid4()),  # 必须是合法 UUID
}

    apply_teacher_decision_node 会按 question_id 找到对应题目，用 new_score 覆盖 final_score。

    6.11.5 为什么整个流程必须在同一个 asyncio.run() 里¶

    llm_factory.py 在模块导入时创建了一个全局 httpx.AsyncClient 单例，用于所有 LLM 的异步调用。httpx.AsyncClient 的连接池绑定到它第一次被使用时所在的事件循环。

    每次 asyncio.run() 都会创建一个新事件循环，运行完毕后**销毁该循环**。如果把两次 ainvoke() 各自包在独立的 asyncio.run() 里，第二次调用时 httpx.AsyncClient 的连接池还绑在已被销毁的旧循环上，立即报 Connection error.。

    把所有 await 写在同一个 async def main() 里，用唯一的 asyncio.run(main()) 驱动，整个链路共享同一个事件循环，问题消失。

    章节总结¶

    • 线性图：9个节点全部固定边，无条件分支，图结构是最简单的情形
    • HitL 依赖 MemorySaver：interrupt() 后 State 存入 MemorySaver，Command(resume=...) 时按同一 thread_id 恢复
    • 两次 ainvoke() 同一个 config：thread_id 是找回冻结 State 的唯一标识，两次调用必须保持一致
    • 复杂性在节点内：run_three_tracks（三轨并行）和 teacher_review（interrupt 暂停），图拓扑结构本身很简单
